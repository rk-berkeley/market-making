\section{General Setup and Notation}
\label{sec:setup}

Throughout this report, we work on a filtered probability space
$(\Omega, \F, (\F_t)_{t \ge 0}, \Prob)$ satisfying the usual conditions.
We fix a risk-neutral pricing measure $\Q$ and denote by
$W = (W_t)_{t \ge 0}$ a standard Brownian motion under $\Q$.

\begin{definition}[Black-Scholes parameters]
The following parameters are fixed in Section~\ref{sec:asian}:
$S_0 > 0$ (initial asset price), $r \ge 0$ (risk-free rate),
$\sigma > 0$ (volatility), $T > 0$ (maturity), $K > 0$ (strike),
$n \in \mathbb{N}$ (discretisation steps), and $M \in \mathbb{N}$
(Monte Carlo sample size).
\end{definition}

\begin{definition}[Black-Scholes dynamics]
\label{def:BS}
Under $\Q$, the asset price $S = (S_t)_{t \ge 0}$ satisfies
\begin{equation}
  \dd S_t = r S_t \, \dd t + \sigma S_t \, \dd W_t, \quad S_0 > 0.
\end{equation}
By It\^{o}'s formula the unique strong solution is
\begin{equation}
  S_t = S_0 \exp\!\Bigl(\sigma W_t + \bigl(r - \tfrac{\sigma^2}{2}\bigr)t\Bigr).
\end{equation}
\end{definition}

\subsection{The Asian Option}
\label{subsec:asian-def}

\begin{definition}[Asian call option]
An \emph{Asian call option} with maturity $T$ and strike $K$ has payoff
\begin{equation}
  \Psi(A_T) = (A_T - K)^+,
\end{equation}
where $A_T = \frac{1}{T}\int_0^T S_t \, \dd t$ is the continuous-time
arithmetic average of the asset over $[0,T]$.
\end{definition}

\begin{definition}[Discretised averages]
On the uniform grid $t_i = iT/n$, $i = 0,\ldots,n$, we approximate $A_T$ by
\begin{align}
  A_T^{\mathrm{arith}} &:= \frac{1}{n+1}\sum_{i=0}^{n} S_{t_i},
  \label{eq:arith} \\
  A_T^{\mathrm{geom}}  &:= \exp\!\Bigl(\frac{1}{n+1}\sum_{i=0}^{n}\ln S_{t_i}\Bigr).
  \label{eq:geom}
\end{align}
\end{definition}

\begin{definition}[Asian call price]
The no-arbitrage price at time $0$ is
\begin{equation}
  \mathrm{CallAsian}(r,\sigma,S_0,T,K)
  = e^{-rT}\,\E^{\Q}\!\bigl[(A_T - K)^+\bigr].
\end{equation}
\end{definition}

\begin{remark}
Unlike the geometric case, no closed-form price exists for the arithmetic
Asian option in the Black-Scholes model, since the arithmetic average of
log-normals is not log-normal. This motivates Monte Carlo methods.
\end{remark}

\subsection{Question 1 --- Black-Scholes Formula and Convexity}
\label{subsec:q1}

\begin{theorem}[Black-Scholes formula]
\label{thm:bs}
The price at time $0$ of a European call with strike $K$ and maturity $T$ is
\begin{equation}
  \mathrm{CallBS}(S_0,K,r,\sigma,T)
  = S_0\,\Phi(d_1) - K e^{-rT}\,\Phi(d_2),
\end{equation}
where $\Phi$ is the standard normal CDF and
\begin{equation}
  d_1 = \frac{\ln(S_0/K)+(r+\frac{1}{2}\sigma^2)T}{\sigma\sqrt{T}},
  \qquad d_2 = d_1 - \sigma\sqrt{T}.
\end{equation}
\end{theorem}

\begin{property}[Convexity in $K$]
\label{prop:convex}
The map $K \mapsto \mathrm{CallBS}(S_0,K,r,\sigma,T)$ is convex and
decreasing on $(0,\infty)$.
\end{property}

\begin{proof}
Convexity follows because $K \mapsto (S_T - K)^+$ is convex for every
realisation of $S_T$ and the discounted expectation preserves convexity.
Analytically, $\partial^2\mathrm{CallBS}/\partial K^2
= e^{-rT}\phi(d_2)/(\sigma\sqrt{T}\,K) > 0$,
where $\phi$ denotes the standard normal PDF.
\end{proof}

\begin{example}[Numerical convexity check]
With $S_0=50$, $r=0.05$, $\sigma=0.30$, $T=1$ and a uniform strike grid
$K\in\{35,36,\ldots,55\}$ ($h=1$), we compute
$\Delta^2_h C_{\mathrm{BS}}(K) = C_{\mathrm{BS}}(K-1)-2C_{\mathrm{BS}}(K)+C_{\mathrm{BS}}(K+1)$
at each interior point.
Table~\ref{tab:bs-convex} reports the call prices; the minimum second
finite difference is $1.893\times10^{-2}>0$, confirming
Property~\ref{prop:convex} numerically.
\end{example}

\subsection{Question 2 --- Monte Carlo Pricing}

\begin{definition}[Monte Carlo estimator]
Given $M$ i.i.d.\ copies $X^{(1)},\ldots,X^{(M)}$ of a random variable $X$,
the Monte Carlo estimator of $\E[X]$ is
\begin{equation}
  \bar{X}_M := \frac{1}{M}\sum_{j=1}^{M} X^{(j)}
  \xrightarrow{a.s.} \E[X] \quad \text{as } M\to\infty.
\end{equation}
\end{definition}

\begin{remark}[Exact path simulation]
We simulate the asset using the exact log-normal increment:
\[
  S_{t_{i+1}} = S_{t_i}
  \exp\!\Bigl(\bigl(r-\tfrac{\sigma^2}{2}\bigr)\Delta t
              + \sigma\sqrt{\Delta t}\,Z_i\Bigr),
  \quad Z_i \overset{\text{i.i.d.}}{\sim}\mathcal{N}(0,1),
\]
with $\Delta t = T/n$. This introduces no time-discretisation error.
\end{remark}

We set $S_0 = 50$, $n = 50$, $M = 200{,}000$ and explore the parameter

\subsection{Question 3 --- Confidence Intervals}
\label{subsec:q3}

\begin{theorem}[95\% confidence interval]
\label{thm:ci}
Let $s_M^2 = \frac{1}{M}\sum_{j=1}^M |X^{(j)}-\bar{X}_M|^2$.
By the Central Limit Theorem,
\[
  \frac{\bar{X}_M - \E[X]}{s_M/\sqrt{M}} \xrightarrow{\mathcal{L}}
  \mathcal{N}(0,1).
\]
An asymptotic 95\% confidence interval for $\E[X]$ is
\begin{equation}
  \mathrm{CI}_{0.95} =
  \Bigl[\bar{X}_M - 1.96\,\tfrac{s_M}{\sqrt{M}},\;
        \bar{X}_M + 1.96\,\tfrac{s_M}{\sqrt{M}}\Bigr],
  \quad |\mathrm{CI}_{0.95}| = \tfrac{2\times1.96\,s_M}{\sqrt{M}}.
\end{equation}
\end{theorem}

\begin{property}[Geometric mean reduces variance]
By the AM-GM inequality, $A_T^{\mathrm{geom}} \le A_T^{\mathrm{arith}}$
path-wise, hence
\[
  \Var^{\Q}\!\bigl[(A_T^{\mathrm{geom}}-K)^+\bigr]
  \;\le\;
  \Var^{\Q}\!\bigl[(A_T^{\mathrm{arith}}-K)^+\bigr].
\]
Consequently, the 95\% CI for $\mathrm{CallAsian}^{\mathrm{geom}}$ is
strictly shorter than for $\mathrm{CallAsian}^{\mathrm{arith}}$.
\end{property}

Figure~\ref{fig:ci} illustrates this variance reduction across all strikes.
Numerically, for $r=0.05$, $\sigma=0.30$, $T=1$, $M=200{,}000$
(see Table~\ref{tab:mc-prices-1}), the 95\% CI widths are:

\begin{itemize}[noitemsep]
  \item $K=35$: arithmetic $0.0747$, geometric $0.0730$ (reduction $\approx 2.3\%$).
  \item $K=40$: arithmetic $0.0720$, geometric $0.0697$ (reduction $\approx 3.2\%$).
  \item $K=45$: arithmetic $0.0647$, geometric $0.0621$ (reduction $\approx 4.0\%$).
  \item $K=50$: arithmetic $0.0527$, geometric $0.0497$ (reduction $\approx 5.7\%$).
  \item $K=55$: arithmetic $0.0390$, geometric $0.0358$ (reduction $\approx 8.2\%$).
\end{itemize}

The reduction is larger for out-of-the-money strikes, where the geometric
averaging effect is more pronounced.  For longer maturity ($T=2$) and
higher volatility ($\sigma=0.40$), the CI width reduction exceeds $10\%$
(e.g.\ arithmetic $0.1406$ vs.\ geometric $0.1309$ for $r=0$, $\sigma=0.4$,
$K=35$, $T=2$), consistent with Theorem~\ref{thm:ci}.

\subsection{Question 4 --- Comparison with the European Call}
\label{subsec:q4}

\begin{lemma}[AM-GM inequality for averages]
\label{lem:amgm}
For any non-negative real numbers $s_0, s_1, \ldots, s_n > 0$,
\begin{equation}
  \left(\prod_{i=0}^{n} s_i\right)^{\!\frac{1}{n+1}}
  \;\le\;
  \frac{1}{n+1}\sum_{i=0}^{n} s_i.
\end{equation}
In particular, applied path-wise to the discretised asset prices,
$A_T^{\mathrm{geom}} \le A_T^{\mathrm{arith}}$ almost surely.
\end{lemma}

\begin{proof}
The AM-GM inequality follows from the concavity of $\ln$: by Jensen's
inequality applied to the concave function $x \mapsto \ln x$,
\[
  \frac{1}{n+1}\sum_{i=0}^{n}\ln s_i
  \;\le\;
  \ln\!\left(\frac{1}{n+1}\sum_{i=0}^{n} s_i\right).
\]
Exponentiating both sides gives the result.
\end{proof}

\begin{theorem}[Price ordering]
\label{thm:order}
In the Black-Scholes model,
\begin{equation}
  \mathrm{CallAsian}^{\mathrm{geom}}
  \;\le\;
  \mathrm{CallAsian}^{\mathrm{arith}}
  \;\le\;
  \mathrm{CallBS}.
\end{equation}
\end{theorem}

\begin{proof}
The left inequality follows directly from Lemma~\ref{lem:amgm}:
since $A_T^{\mathrm{geom}} \le A_T^{\mathrm{arith}}$ almost surely and
$(x-K)^+$ is non-decreasing, taking expectations under $\Q$ gives
$\mathrm{CallAsian}^{\mathrm{geom}} \le \mathrm{CallAsian}^{\mathrm{arith}}$.

For the right inequality, note that $(x - K)^+$ is convex in $x$.
Since $A_T^{\mathrm{arith}} = \frac{1}{T}\int_0^T S_t \, \dd t$,
Jensen's inequality applied to the convex function $x \mapsto (x-K)^+$ gives
\[
  \left(\frac{1}{T}\int_0^T S_t \, \dd t - K\right)^+
  \;\le\;
  \frac{1}{T}\int_0^T (S_t - K)^+ \, \dd t.
\]
Taking expectations under $\Q$ and discounting,
\[
  \mathrm{CallAsian}^{\mathrm{arith}}
  = e^{-rT}\,\E^{\Q}\!\left[\left(A_T^{\mathrm{arith}} - K\right)^+\right]
  \;\le\;
  \frac{e^{-rT}}{T}\int_0^T \E^{\Q}\!\left[(S_t - K)^+\right] \dd t.
\]
Since $t \mapsto e^{-rt}\,\E^{\Q}[(S_t - K)^+]$ is the price of a European
call with maturity $t$, and the Black-Scholes call price is increasing in
time to maturity (positive theta), we have
$\E^{\Q}[(S_t - K)^+] \le e^{r(T-t)}\,\E^{\Q}[(S_T - K)^+]$ for all
$t \le T$. Therefore,
\[
  \frac{e^{-rT}}{T}\int_0^T \E^{\Q}\!\left[(S_t - K)^+\right] \dd t
  \;\le\;
  e^{-rT}\,\E^{\Q}\!\left[(S_T - K)^+\right]
  = \mathrm{CallBS},
\]
which completes the proof.
\end{proof}

\begin{remark}
Intuitively, averaging smooths out extreme values of $S$, reducing the
volatility of the payoff and hence its price relative to the European call.
\end{remark}

The ordering is confirmed numerically in Tables~\ref{tab:mc-prices-1}
and~\ref{tab:mc-prices-2}, and illustrated in Figure~\ref{fig:compare}.
Several key observations emerge from the full results:

\begin{enumerate}[noitemsep, label=(\roman*)]
  \item \textbf{Price ordering.}
    The inequality $C^g_{\mathrm{Asian}} \le C^a_{\mathrm{Asian}} \le C_{\mathrm{BS}}$
    holds for all 100 parameter combinations tested.
    For example, at $r=0.05$, $\sigma=0.30$, $T=1$, $K=50$:
    $C^g = 3.722 < C^a = 3.957 < C_{\mathrm{BS}} = 7.116$.

  \item \textbf{Volatility effect.}
    All prices increase with $\sigma$. The gap between Asian and European
    prices widens: at $K=40$, $T=1$, $r=0$, the difference
    $C_{\mathrm{BS}} - C^a$ grows from $0.561$ ($\sigma=0.20$) to
    $2.344$ ($\sigma=0.40$), as averaging smooths out more volatility.

  \item \textbf{Maturity effect.}
    Longer maturities yield higher prices. At $r=0.05$, $\sigma=0.30$,
    $K=40$: $C^a$ increases from $10.961$ ($T=1$) to $12.031$ ($T=2$),
    and $C_{\mathrm{BS}}$ from $13.231$ to $16.097$.

  \item \textbf{High interest rate.}
    At $r=1.00$, the European call substantially exceeds the Asian calls

    reflecting the large discounting effect and the fact that the average
    is taken over the entire path.
\end{enumerate}

\section{Specific Topic: Optimal Market-Making under Fad Price Dynamics}
\label{sec:specific}

This section is based on \cite{barucci2025market}.

% ----------------------------------------------------------
\subsection{Introduction}
\label{subsec:intro}

\subsubsection{Problem Overview}

This section examines an optimal market-making problem where asset prices
may temporarily deviate from their fundamental value because of short-term
``fad'' effects. In this setting, a market maker continuously posts bid and
ask quotes, earns the bid--ask spread when orders arrive, and manages the
risk associated with holding inventory. The main goal is to determine
optimal quote displacements that maximise expected terminal wealth, while
still penalising inventory risk.

Unlike the standard Avellaneda--Stoikov setup, this model includes both
uninformed and informed traders. Uninformed traders generate
liquidity demand in a random way, while informed traders respond to
temporary price deviations and create adverse-selection risk. As a result,
the market maker's optimal quotes depend not only on current inventory, but
also on the fad signal and, importantly, on whether that signal is
observable.

\subsubsection{Literature Review}

This work follows the classical stochastic-control approach to market
making \cite{avellaneda2008high}. In this framework, a dealer posts bid and
ask quotes, takes on inventory risk from a diffusive mid-price, and receives
market orders through Poisson arrival processes. Later studies extend this
setup by adding inventory constraints and showing that the resulting
Hamilton--Jacobi--Bellman (HJB) system can be converted into more tractable
ordinary differential equations \cite{gueant2013dealing}.

The paper also relates to the textbook by Cartea, Jaimungal and Penalva
\cite{cartea2015algorithmic}, which covers market making, optimal execution
and informed trading using the same stochastic-control tools. Compared with
those classical models, the key difference here is the mean-reverting fad
component, which enters both the price SDE and the order arrival rates.
This is what generates the distinction between informed and uninformed
traders and creates the adverse-selection problem the market maker faces.

\subsubsection{Main Challenges and Need for Stochastic Calculus}

Stochastic calculus is needed here because the value function depends on
two types of randomness at the same time: continuous diffusion dynamics
from the fad $U_t$, and discrete jumps from the Poisson order arrivals
$N^a_t$ and $N^b_t$. The optimal control problem is handled through an
HJB equation. In the partial-information case, the fad cannot be observed
directly, so filtering theory is used to replace it with its conditional
expectation given the observed price.

\subsubsection{Techniques and Main Results}

The problem is formulated as a stochastic control problem and analysed
using the HJB framework. Because the exact HJB equation involves nonlinear
jump terms, inventory constraints, and clipping terms, closed-form solutions
are not available in general. To address this, the value function is
characterised as a viscosity solution, and a tractable approximate
full-information solution is developed.

For tractability, the approximation removes binding constraints and applies
Taylor expansions to the exponential terms, yielding a quadratic
ansatz for the value function:
\[
  V(t,q,u) = q^2 A(t) + q B(t,u) + C(t,u).
\]
This transformation reduces the HJB equation to a Riccati equation for
$A(t)$, while the remaining coefficients $B$ and $C$ satisfy linear ODEs.

In the partial-information setting, the fad is not directly observable, so
it is replaced by its filtered estimate $\hat{U}_t$. The overall structure
of the optimal quotes stays the same, but the signal variable is now
replaced by this estimate. The main qualitative finding is that, in this
model, adverse selection from informed traders is channelled entirely
through quote skew (the asymmetry $\delta^{a,*}-\delta^{b,*}$),
while the bid--ask spread $\delta^{a,*}+\delta^{b,*}$ remains invariant in
the share of informed traders, unlike the classical
Glosten--Milgrom intuition. Better information about the fad sharpens the
skew response and raises the mean objective, with the full-information
strategy outperforming both the partial-information and
Cartea--Jaimungal--Penalva benchmarks at our parameters.

\subsection{Mathematical and Model Setup}
\label{subsec:model}

\subsubsection{Probability Framework and Notation}

Fix a finite trading horizon $\mathcal{T} = [0,T]$ with $T>0$.
We work on a filtered probability space
$(\Omega, \F, (\F_t)_{t\in\mathcal{T}}, \Prob)$ that supports four
mutually independent objects:
\begin{itemize}[noitemsep]
  \item two one-dimensional standard Brownian motions $(B_t)_{t\in\mathcal{T}}$
        and $(Z_t)_{t\in\mathcal{T}}$;
  \item two unit-rate Poisson processes $(\bar{N}^a_t)_{t\in\mathcal{T}}$ and
        $(\bar{N}^b_t)_{t\in\mathcal{T}}$ (under a reference measure
        $\Prob_0$).
\end{itemize}
We take $(\F_t)_{t\in\mathcal{T}}$ to be the right-continuous, completed
filtration jointly generated by these four processes. The controlled point
processes $(N^a_t)_{t\in\mathcal{T}}$ and $(N^b_t)_{t\in\mathcal{T}}$ that
appear throughout this section are obtained from $(\bar{N}^{a,b})$ via an
absolutely continuous change of measure (a Doléans--Dade exponential), as in
\cite{barucci2025market}; under the resulting measure $\Prob^{\delta}$,
they become point processes whose stochastic intensities depend on the
state in the way specified below. Expectations are taken under
$\Prob^{\delta}$ unless we say otherwise, and we write this simply as $\E$.

\paragraph{Notation.} Throughout, $t\in\mathcal{T}$ is the time index.
A hat (e.g.\ $\hat{U}_t$) means the conditional expectation given a
sub-filtration, and $a,b$ label the ask and bid sides of the order book.

\subsubsection{Price and Fad Dynamics}

The mid-price $(S_t)_{t\in\mathcal{T}}$ and the latent fad
$(U_t)_{t\in\mathcal{T}}$ are both exogenous diffusions, taken from
\cite{barucci2025market}. The mid-price follows
\begin{equation}
\label{eq:dS}
  \dd S_t = \mu \,\dd t + \sigma \,\dd W_t, \qquad S_0 \in \R_+,
\end{equation}
where $\mu\in\R$ and $\sigma>0$. The driving Brownian motion is split
into two pieces,
\begin{equation}
\label{eq:Wsplit}
  W_t = p\,Z_t + q\,U_t, \qquad p,q>0,\ \ p^2+q^2=1.
\end{equation}
so that $(Z_t)$ carries the classical fundamental information flow and
$(U_t)$ stands for the behaviour-driven, transient component of the price.
The latter is itself a mean-reverting Ornstein--Uhlenbeck (OU) process,
\begin{equation}
\label{eq:dU}
  \dd U_t = -\eta\,U_t\,\dd t + \dd B_t, \qquad U_0=0,
\end{equation}
with mean-reversion speed $\eta>0$. The weight $q\in(0,1)$ tunes how
much of the price variability is due to the fad: a larger $q$ means a
noisier mid-price relative to its fundamental value.

\begin{remark}[Fundamental price]
\label{rem:fundamental}
Define the fundamental price
\(F_t := S_t - \sigma q U_t.\)
Substituting \eqref{eq:Wsplit}--\eqref{eq:dU} into \eqref{eq:dS} and using
$\dd U_t=-\eta U_t\,\dd t + \dd B_t$, one obtains the alternative
representation
\[
  \dd S_t = (\mu - \eta\sigma q U_t)\,\dd t + \sigma \,\dd \bar{W}_t,
  \qquad \bar{W}_t = pZ_t + qB_t,
\]
where $\bar{W}$ is a Brownian motion correlated with $B$ through
$\langle \bar{W},B\rangle_t = qt$.
\end{remark}

\subsubsection{Market Maker State Dynamics}

The market maker posts unit-size limit orders on both sides of the book at
all times. Her two controls are the bid and ask displacements
\(\delta^a_t,\delta^b_t \ge -\delta_\infty,\) and from these she quotes
the prices
\begin{equation}
\label{eq:quotes}
  S^a_t = S_t + \delta^a_t \quad\text{(ask)}, \qquad
  S^b_t = S_t - \delta^b_t \quad\text{(bid)}.
\end{equation}
A filled bid grows her inventory by one unit; a filled ask shrinks it by
one. So the inventory and cash processes evolve as
\begin{align}
  \dd Q_t &= \dd N^b_t - \dd N^a_t, & Q_0&\in\mathcal{Q},
  \label{eq:dQ}\\
  \dd X_t &= (S_t+\delta^a_t)\,\dd N^a_t - (S_t-\delta^b_t)\,\dd N^b_t,
  & X_0&=0.
  \label{eq:dX}
\end{align}
Inventory cannot grow without bound: we impose hard limits
$\underline{q}<0<\bar{q}$ and require
$Q_t \in \mathcal{Q}=\{\underline{q},\ldots,\bar{q}\}\subset\mathbb{Z}$.
Whenever a fill would push inventory past the limit on one side, the market
maker pulls her quote on that side rather than trade, a constraint encoded
by the indicators in the intensity formulas below.

\subsubsection{Order Arrival Intensities}

Order flow comes from two distinct populations of traders:
uninformed traders, who only observe the mid-price, and
informed traders, who can read the fad off the price and trade
against the fundamental. The market maker sees only the aggregate,
anonymous flow and cannot tell the two apart at the level of individual
trades.

The stochastic intensities of the controlled counting processes $(N^a_t)$
and $(N^b_t)$ are
\begin{align}
  \lambda^a_t &= \Bigl(\underbrace{\phi\, e^{-k\delta^a_t}}_{\text{uninformed}}
    \;+\; \underbrace{\psi\, e^{-k\delta^a_t - \gamma(\sigma q U_t \vee S^-)}}_{\text{informed}}\Bigr)
    \mathbf{1}_{\{Q_{t-} > \underline{q}\}},
  \label{eq:lambda-a}\\
  \lambda^b_t &= \Bigl(\underbrace{\phi\, e^{-k\delta^b_t}}_{\text{uninformed}}
    \;+\; \underbrace{\psi\, e^{-k\delta^b_t + \gamma(\sigma q U_t \wedge S^+)}}_{\text{informed}}\Bigr)
    \mathbf{1}_{\{Q_{t-} < \bar{q}\}},
  \label{eq:lambda-b}
\end{align}
where $k,\gamma>0$, $\phi,\psi\ge 0$, and $S^-\le 0\le S^+$ are the
clipping levels. Capping the toxic term at $\sigma q U_t \vee S^-$ on the
ask side (and similarly at $\sigma q U_t \wedge S^+$ on the bid) keeps the
intensities bounded, which is what we need for the Doléans--Dade exponential
in the construction of $\Prob^{\delta}$ to be a true martingale.

\paragraph{Economic interpretation.}
Uninformed traders only see the gap between the quote and the mid-price
($\delta^a_t$ on the ask, $\delta^b_t$ on the bid), so their arrival rate
decays exponentially in the displacement at rate $k$. Informed traders, by
contrast, can observe the fundamental $F_t = S_t-\sigma q U_t$ and react
to the gap between the quote and that: from their point of view,
the ``generosity'' of the ask is $S^a_t-F_t = \delta^a_t + \sigma q U_t$
and that of the bid is $F_t-S^b_t = \delta^b_t - \sigma q U_t$. A positive
fad signals an over-priced asset and pulls in informed sellers; a negative
one signals the opposite and brings in informed buyers. This is the channel
through which adverse selection enters the market maker's problem. The
parameter $\gamma$ controls how aggressively informed flow responds to the
fad, while the ratio $\psi/\phi$ sets the relative size of the two
populations.

\subsubsection{Objective Function}

The market maker is risk-averse in inventory and wants to maximise the
performance criterion from \cite{cartea2015algorithmic},
\begin{equation}
\label{eq:objective}
  J(\delta) \;=\; \E\!\left[\,X_T + Q_T S_T - \alpha\,Q_T^2
        - \varphi\!\int_0^T Q_u^2\,\dd u \,\right],
\end{equation}
over admissible controls $\delta=(\delta^a,\delta^b)$. Each term in
\eqref{eq:objective} carries a clear economic meaning:
\begin{itemize}[noitemsep]
  \item $X_T$ is the realised cash position at terminal time $T$;
  \item $Q_T S_T$ is the mark-to-market value of any leftover inventory at
        the prevailing mid-price (we do not allow the market maker to
        recover the fundamental component when she liquidates);
  \item $\alpha Q_T^2$ penalises non-zero terminal inventory; this is equivalent to a linear-in-size liquidation slippage, since
        $Q_T S_T - \alpha Q_T^2 = Q_T(S_T - \alpha Q_T)$;
  \item $\varphi\int_0^T Q_u^2\,\dd u$ is a running inventory penalty
        that pushes the market maker towards a flat book throughout the
        horizon.
\end{itemize}
The set of admissible controls is
\[
  \mathcal{U} \;:=\;
  \bigl\{\delta=(\delta^a,\delta^b)\in L^2(\Omega\times[0,T])\;:\;
        \delta\ge -\delta_\infty,\ \F\text{-predictable}\bigr\}.
\]

% ----------------------------------------------------------
\subsection{Optimisation Problem and Relevant Quantities}
\label{subsec:optim}

\subsubsection{Controlled SDE System}

Putting together \eqref{eq:dS}, \eqref{eq:dU}, \eqref{eq:dQ}--\eqref{eq:dX}
and the intensities \eqref{eq:lambda-a}--\eqref{eq:lambda-b}, the full
controlled state $\mathbf{X}_t = (X_t,Q_t,S_t,U_t)$ obeys
\begin{equation}
\label{eq:state-system}
  \begin{cases}
    \dd X_t = (S_t+\delta^a_t)\,\dd N^a_t - (S_t-\delta^b_t)\,\dd N^b_t,
        & X_0 = 0,\\[2pt]
    \dd Q_t = \dd N^b_t - \dd N^a_t,
        & Q_0 \in \mathcal{Q},\\[2pt]
    \dd S_t = \mu\,\dd t + \sigma(p\,\dd Z_t + q\,\dd U_t),
        & S_0 \in \R_+,\\[2pt]
    \dd U_t = -\eta\,U_t\,\dd t + \dd B_t,
        & U_0 = 0.
  \end{cases}
\end{equation}
Note that we do not include $Z_t$ in the state, since it is recoverable
from $(S_t,U_t)$ via $Z_t = \sigma^{-1}p^{-1}(S_t-S_0-\mu t-\sigma q U_t)$.
Two of the components ($S$ and $U$) are diffusions and two ($N^a$, $N^b$)
are controlled point processes, so the It\^o formula we will need below
has both Brownian and jump contributions.

\subsubsection{The Stochastic Control Problem}

The market maker's task is to pick $\delta\in\mathcal{U}$ that maximises
the criterion \eqref{eq:objective}. Defining the value function
\begin{equation}
\label{eq:value-fn}
  V(t,q,u) \;:=\; \sup_{\delta\in\mathcal{U}}\,
  \E_{t,q,u}\!\left[\,-\alpha Q_T^2
    + \!\int_t^T\!\!\bigl(\delta^a_s\lambda^a_s + \delta^b_s\lambda^b_s
      - \varphi Q_s^2 + Q_s\mu - \eta q\sigma Q_s U_s\bigr)\,\dd s\right],
\end{equation}
the original problem $\sup_\delta J(\delta)$ becomes equivalent to
evaluating $V(0,Q_0,0)$, after the It\^o reduction we carry out in
Section~\ref{subsec:solutions}. We use $\E_{t,q,u}$ as shorthand for the
conditional expectation given $Q_t=q$ and $U_t=u$.

\subsubsection{Relevant Quantities of Interest}

The main things we look at are:
\begin{itemize}[noitemsep]
  \item the optimal feedback controls $\delta^{a,*}(t,q,u)$ and
        $\delta^{b,*}(t,q,u)$, written as finite differences of $V$ in $q$;
  \item the value function $V(t,q,u)$ itself, which under the quadratic
        ansatz of Section~\ref{subsec:solutions} collapses to three
        deterministic functions $A(t)$, $B(t,u)$, and $C(t,u)$;
  \item the equilibrium bid--ask spread $\delta^{a,*}+\delta^{b,*}$ and
        how it varies with the model parameters $(q,\gamma,\eta,\phi,\psi)$;
  \item the inventory and cash trajectories that the optimal policy
        produces, together with the resulting profit-and-loss distribution.
\end{itemize}

\subsubsection{Information Structure}

One important feature of this model is that the same problem can be run
under different assumptions about what the market maker can observe. We
compare three cases following \cite{barucci2025market}:

\begin{enumerate}[label=(\roman*),noitemsep]
  \item \textbf{Full information (FI).} The market maker observes the
        entire filtration $\F = (\F_t)_{t\in\mathcal{T}}$, and in
        particular the realised fad path $(U_t)$. Her optimal control is a
        feedback function of $(t,Q_{t-},U_t)$.

  \item \textbf{Partial information (PI).} The market maker only sees the
        price filtration $\F^S = (\F^S_t)_{t\in\mathcal{T}}$ generated by
        $(S_t)$ (and the order flow). The fad $(U_t)$ is hidden, and the
        best she can do is replace it by its conditional expectation
        $\hat{U}_t = \E[U_t\mid\F^S_t]$, computed by a Kalman--Bucy
        filter. Her optimal control is then a feedback function of
        $(t,Q_{t-},\hat{U}_t)$.

  \item \textbf{CJP benchmark.} The market maker uses the
        Cartea--Jaimungal--Penalva strategy
        \cite[Chapter 10]{cartea2015algorithmic}, which ignores the fad
        entirely (it assumes $q=0$ and $\gamma=0$). Her control depends
        only on $(t,Q_{t-})$. The actual market does have a fad, so the
        CJP strategy is computed under a misspecified model and serves as
        a natural baseline.
\end{enumerate}

Comparing these three strategies in Section~\ref{subsec:numerics} tells
us about the value of information: the gap between FI and PI tells us
what the market maker loses by not seeing the fad directly, and the gap
between PI and CJP tells us how much it pays to model the fad correctly
even when it cannot be observed.

% ----------------------------------------------------------
\subsection{Solutions}
\label{subsec:solutions}

Following \cite{barucci2025market}, maximising $J(\delta)$ is equivalent to
maximising the value function $V(t,q,u)$ defined in \eqref{eq:value-fn},
up to the constant $Q_0 S_0$. We work with $V$ directly.

\subsubsection{Dynamic Programming and the HJB Equation}

By the standard dynamic programming principle for jump-diffusion control
problems \cite[Ch.\ 5]{shreve2004stochastic}, the value function $V$
formally satisfies a Hamilton--Jacobi--Bellman (HJB) equation. The
infinitesimal generator of the state splits naturally into two pieces: a
diffusion part inherited from $U$, and a jump part inherited from $N^a$
and $N^b$.

\begin{theorem}[HJB equation]
\label{thm:hjb}
The value function $V(t,q,u)$ defined in \eqref{eq:value-fn} formally
solves the HJB equation
\begin{equation}
\label{eq:hjb}
\begin{aligned}
  0 \;=\;& \partial_t V - \eta u\,\partial_u V
    + \tfrac{1}{2}\partial^2_{uu}V
    - \varphi q^2 + q(\mu - \eta\sigma q u)\\
   &+ \sup_{\delta^a}\Bigl\{
        e^{-k\delta^a}\bigl(\phi + \psi e^{-\gamma(\sigma q u\vee S^-)}\bigr)
        \bigl[\delta^a + V(t,q-1,u) - V(t,q,u)\bigr]\Bigr\}
        \mathbf{1}_{\{q>\underline{q}\}}\\
   &+ \sup_{\delta^b}\Bigl\{
        e^{-k\delta^b}\bigl(\phi + \psi e^{\gamma(\sigma q u\wedge S^+)}\bigr)
        \bigl[\delta^b + V(t,q+1,u) - V(t,q,u)\bigr]\Bigr\}
        \mathbf{1}_{\{q<\bar{q}\}},
\end{aligned}
\end{equation}
with terminal condition $V(T,q,u) = -\alpha q^2$.
\end{theorem}

The four lines of \eqref{eq:hjb} read as follows. The first line collects
the deterministic drift and the running inventory penalty; the second line
is the OU diffusion generator acting in $u$; and the last two ``$\sup$''
terms are the jump contributions on the ask and bid sides, where the
inventory $q$ jumps by $\pm 1$ each time an order is filled.

\subsubsection{Optimal Quote Displacements}

Each supremum in \eqref{eq:hjb} has the same generic form,
$\max_{\delta} e^{-k\delta}(\delta + V_*)$, where $V_*$ is a finite
difference of the value function in $q$. The first-order condition gives
\[
  \tfrac{\dd}{\dd \delta}\bigl[e^{-k\delta}(\delta + V_*)\bigr]
  = e^{-k\delta}\bigl[1 - k(\delta + V_*)\bigr] = 0
  \;\Longrightarrow\; \delta = \frac{1}{k} - V_*.
\]

\begin{proposition}[Optimal feedback controls]
\label{prop:optimal-controls}
Whenever the lower bound $-\delta_\infty$ is non-binding, the unique
maximisers of \eqref{eq:hjb} are
\begin{align}
  \delta^{a,*}(t,q,u) &= \frac{1}{k} - \bigl[V(t,q-1,u) - V(t,q,u)\bigr],
        \qquad q\ne\underline{q},
        \label{eq:optimal-a}\\
  \delta^{b,*}(t,q,u) &= \frac{1}{k} - \bigl[V(t,q+1,u) - V(t,q,u)\bigr],
        \qquad q\ne\bar{q}.
        \label{eq:optimal-b}
\end{align}
The constraint is enforced by capping at $-\delta_\infty$ if necessary.
\end{proposition}

The optimal displacement on each side is the elasticity
term $1/k$, minus the marginal value of holding one extra unit of
inventory in the appropriate direction.

\subsubsection{Viscosity Solution Characterisation}

The HJB equation \eqref{eq:hjb} is a non-local, non-smooth PDE: the jump
terms link $V$ at $(t,q,u)$ to $V$ at $(t,q\pm 1,u)$, and the
inventory-bound indicators bring discontinuities into $q$. As a result,
$V$ cannot generally be a classical $C^{1,2}$ solution, and we have to
work with the weaker notion of a viscosity solution
\cite[Ch.\ 5]{touzi2012optimal}.

Let $V_*(t,q,u) = \liminf V$ and $V^*(t,q,u) = \limsup V$ denote the
lower and upper semicontinuous envelopes (limits taken at $(t,q,u)$), and
write the Hamiltonian as
\begin{equation}
\label{eq:hamiltonian}
\begin{aligned}
  H(q,u,V_1,V_2) \;:=\;
   &\sup_{\delta^a}\bigl(\phi + \psi e^{-\gamma(\sigma q u\vee S^-)}\bigr)
      e^{-k\delta^a}(\delta^a+V_1)\,\mathbf{1}_{\{q>\underline{q}\}}\\
   &+ \sup_{\delta^b}\bigl(\phi + \psi e^{\gamma(\sigma q u\wedge S^+)}\bigr)
      e^{-k\delta^b}(\delta^b+V_2)\,\mathbf{1}_{\{q<\bar{q}\}}.
\end{aligned}
\end{equation}

\begin{theorem}[Viscosity characterisation]
\label{thm:viscosity}
The value function $V$ is locally bounded on $E:=[0,T]\times\mathcal{Q}\times\R$,
and is a viscosity solution of \eqref{eq:hjb} in the sense that:
\begin{itemize}[noitemsep]
  \item (Sub-solution.) For every $\Phi\in C^{1,2}$ such that
        $V^*-\Phi$ has a local maximum at $(t_0,q_0,u_0)$ with value zero,
        \(F(t_0,q_0,u_0,\Phi,\partial_t\Phi,\partial_u\Phi,\partial^2_{uu}\Phi)\ge 0.\)
  \item (Super-solution.) For every $\Phi\in C^{1,2}$ such that
        $V_*-\Phi$ has a local minimum at $(t_0,q_0,u_0)$ with value zero,
        \(F(t_0,q_0,u_0,\Phi,\partial_t\Phi,\partial_u\Phi,\partial^2_{uu}\Phi)\le 0.\)
\end{itemize}
Here $F$ is the functional given by the right-hand side of
\eqref{eq:hjb} written compactly using $H$.
\end{theorem}



\subsubsection{Approximate Full-Information Solution}

There is no closed-form solution for the HJB equation \eqref{eq:hjb}, so
we follow \cite{bergault2018closed,barucci2025market} and develop a
tractable approximation. The plan is to:
\begin{enumerate}[label=(\roman*),noitemsep]
  \item drop the inventory bounds ($\underline{q}=-\infty$,
        $\bar{q}=+\infty$) and the lower bound on displacements
        ($\delta_\infty=\infty$);
  \item drop the clipping ($S^-=-\infty$, $S^+=+\infty$);
  \item Taylor-expand $p\mapsto k^{-1}e^{kp}$ to second order around zero,
        applied to the finite differences
        $\Delta^\pm V := V(t,q\pm 1,u)-V(t,q,u)$;
  \item Taylor-expand $u\mapsto e^{\pm\gamma\sigma q u}$ to first order
        around zero (valid for $\gamma\sigma q u$ small).
\end{enumerate}

After these four reductions the HJB equation becomes a smooth polynomial
PDE that admits a quadratic-in-$q$ ansatz.

\begin{proposition}[Quadratic ansatz]
\label{prop:quad-ansatz}
Suppose
\(V(t,q,u) = q^2 A(t) + q B(t,u) + C(t,u),\)
where $A:[0,T]\to\R$ and $B,C:[0,T]\times\R\to\R$ solve the system
\begin{equation}
\label{eq:ABC-system}
\begin{cases}
  \partial_t A - \varphi + 4(\phi+\psi)e^{-1}k\,A^2 = 0, \\[2pt]
  \partial_t B + \tfrac12\partial^2_{uu}B + (\mu-\eta\sigma q u)
     - \eta u\,\partial_u B + 4(\phi+\psi)e^{-1}k\,A\,B \\
  \qquad + 4e^{-1}\psi q\sigma\gamma u\,A
     + 4 e^{-1} k q\gamma\sigma u\,\psi\,A^2 = 0, \\[2pt]
  \partial_t C + \tfrac12\partial^2_{uu}C - \eta u\,\partial_u C
     + e^{-1}k^{-1}\!\bigl(2(\phi+\psi) + 2kA(\phi+\psi)
        + 2\psi k B q\sigma\gamma u\\
  \qquad + k^2(\phi+\psi)(A^2+B^2) + 2k^2\psi q\sigma\gamma u\,A B\bigr) = 0,
\end{cases}
\end{equation}
with terminal conditions $A(T)=-\alpha$, $B(T,u)=0$, $C(T,u)=0$. Then $V$
solves the approximate HJB equation.
\end{proposition}

The first line of \eqref{eq:ABC-system} is a scalar Riccati ODE for
$A(t)$. Setting $\kappa := 4(\phi+\psi)e^{-1}k$, its closed-form solution
(in the generic case $\sqrt{\varphi}\ne\sqrt{\kappa}\alpha$) is
\begin{equation}
\label{eq:A-explicit}
  A(t) \;=\; \sqrt{\varphi/\kappa}\,
    \frac{1-e^{2\sqrt{\varphi\kappa}(T-t)}\,\beta}
         {1+e^{2\sqrt{\varphi\kappa}(T-t)}\,\beta},
  \qquad \beta = \frac{\sqrt{\varphi}+\sqrt{\kappa}\alpha}
                       {\sqrt{\varphi}-\sqrt{\kappa}\alpha}.
\end{equation}

\begin{proposition}[Reduction to a linear ODE system]
\label{prop:linear-odes}
Postulate $B(t,u) = b_0(t)+u\,b_1(t)$ and $C(t,u)=c_0(t)+u\,c_1(t)+u^2 c_2(t)$.
Then $(b_0,b_1,c_0,c_1,c_2)$ solve a linear ODE system with terminal
condition zero. The system can be solved by the Duhamel formula and
straightforward numerical integration.
\end{proposition}

Plugging the quadratic ansatz back into \eqref{eq:optimal-a}--\eqref{eq:optimal-b}
gives the optimal feedback controls in compact form:
\begin{equation}
\label{eq:optimal-quadratic}
  \delta^{a,*}(t,q,u) \;=\; \frac{1}{k} + (2q-1)A(t) + B(t,u),
  \qquad
  \delta^{b,*}(t,q,u) \;=\; \frac{1}{k} - (2q+1)A(t) - B(t,u).
\end{equation}
Two consequences are immediate:
\begin{itemize}[noitemsep]
  \item the optimal spread,
        $\delta^{a,*}+\delta^{b,*} = 2/k - 2A(t)$, depends only on
        $A(t)$ and therefore only on the structural parameters
        $(\varphi,\phi,\psi,k,\alpha)$, never on the fad $u$. This is
        what explains the observation in Figure~8b of
        \cite{barucci2025market} that the spread is invariant in the
        share of informed traders;
  \item the asymmetry (or skew) between the two quotes is
        $\delta^{a,*}-\delta^{b,*} = 2(qA(t)+B(t,u))$, which carries all
        of the dependence on inventory $q$ and on the fad signal $u$. So
        in the FI strategy, inventory and signal information enter only
        through the skew, never through the spread.
\end{itemize}

\subsubsection{Partial-Information Solution via Filtering}

In the partial-information regime the market maker no longer observes
$(U_t)$ directly. She has to infer it from the price filtration
$\F^S$. We follow the classical Kalman--Bucy paradigm for linear-Gaussian
filtering \cite{bain2009fundamentals}, and define the conditional mean and
conditional variance
\begin{equation}
\label{eq:hatU}
  \hat{U}_t := \E[U_t\mid\F^S_t], \qquad
  \hat{P}_t := \E[(U_t-\hat{U}_t)^2 \mid \F^S_t].
\end{equation}
Define the innovation process
\[
  \dd I_t = \dd S_t - \E[\mu - \eta\sigma q U_t \mid \F^S_t]\,\dd t,
\]
which, after rescaling by $\sigma^{-1}$, is an $\F^S$-Brownian motion
\cite{bain2009fundamentals}.

\begin{theorem}[Kalman--Bucy filter for the fad]
\label{thm:kalman}
The conditional mean $\hat{U}_t$ satisfies the SDE
\begin{equation}
\label{eq:hatU-dyn}
  \dd \hat{U}_t = -\eta\,\hat{U}_t\,\dd t
    + \sigma^{-1}\bigl(-\eta q\,\hat{P}_t + q\bigr)\,\dd I_t,
  \qquad \hat{U}_0 = 0.
\end{equation}
The conditional variance $\hat{P}_t$ is deterministic and solves the
Riccati ODE
\begin{equation}
\label{eq:hatP-dyn}
  \tfrac{\dd}{\dd t}\hat{P}_t
  \;=\; -\eta^2 q^2\,\hat{P}_t^2
       - \hat{P}_t(2\eta - 2\eta q^2) + p^2,
  \qquad \hat{P}_0 = 0.
\end{equation}
\end{theorem}



Once we have the filter, the partial-information control problem is
formally identical to the full-information one, with $U_t$ replaced by
$\hat{U}_t$ throughout. The market maker now solves
\begin{equation}
\label{eq:value-pi}
  V^{\mathrm{PI}}(t,q,\hat{u}) \;=\;
  \sup_{\delta\in\mathcal{U}^S}
  \E_{t,q,\hat{u}}\!\left[
    -\alpha Q_T^2 + \!\int_t^T\!\!\bigl(\delta^a_s\lambda^a_s+\delta^b_s\lambda^b_s
      - \varphi Q_s^2 + (\mu-\eta q\sigma\hat{U}_s)Q_s\bigr)\,\dd s\right],
\end{equation}
where $\mathcal{U}^S$ is the set of admissible controls predictable with
respect to the enlarged filtration generated by
$(\tilde{I}_t,\hat{U}_t,N^{a,b}_t)$, and the intensities are now given by
\cite{barucci2025market}
\begin{align*}
  \lambda^a_t &= (\phi + \psi e^{-\gamma(\sigma q\hat{U}_t\vee S^-)})
        e^{-k\delta^a_t}\mathbf{1}_{\{Q_{t-}>\underline{q}\}},\\
  \lambda^b_t &= (\phi + \psi e^{\gamma(\sigma q\hat{U}_t\wedge S^+)})
        e^{-k\delta^b_t}\mathbf{1}_{\{Q_{t-}<\bar{q}\}}.
\end{align*}
The corresponding HJB equation has the same algebraic structure as
\eqref{eq:hjb}, except that the second-order term in $\hat{u}$ now carries
the diffusion coefficient $(\hat{P}_t q\eta - q)^2$. Since the volatility
coefficients do not enter the linear or quadratic-in-$q$ terms of the value
function, the optimal feedback controls under partial information have
exactly the same form as in \eqref{eq:optimal-quadratic}, with $u$
replaced by $\hat{u}$ and the ODE coefficients $(A,B)$ adjusted
accordingly:
\begin{equation}
\label{eq:optimal-PI}
  \delta^{a,*\mathrm{PI}}(t,q,\hat{u}) = \frac{1}{k} + (2q-1)A(t) + B(t,\hat{u}),
  \qquad
  \delta^{b,*\mathrm{PI}}(t,q,\hat{u}) = \frac{1}{k} - (2q+1)A(t) - B(t,\hat{u}).
\end{equation}
In short, the separation principle says: estimate the hidden state first,
then act as if that estimate were the true state. As \cite{barucci2025market} points out, this is not in general
optimal for non-linear-quadratic problems. In our simulations the PI
mean objective is $J_{\mathrm{PI}}=21.20$ at baseline ($q=0.6$), against
$J_{\mathrm{FI}}=21.33$ (Table~\ref{tab:perf-summary}), so the loss from
substituting the filter for the unobserved $U_t$ is $0.13$ in absolute
units, or about $0.6\%$ of the FI objective. Figure~\ref{fig:perf-q}
shows that this gap shrinks further as $q$ increases and effectively
vanishes at $q=1$, where Proposition~22 of \cite{barucci2025market}
gives $\hat{U}_t = U_t$ exactly.

\subsubsection{Strategy Comparison}

We benchmark the three strategies introduced in
Section~\ref{subsec:optim} (FI, PI, CJP) on simulated paths of the full
model. Baseline parameters follow \cite{barucci2025market}: $T=1$,
$S_0=100$, $X_0=Q_0=0$, $\sigma=1$, $\eta=10$, $q=0.6$ (so
$p=\sqrt{1-q^2}\approx 0.8$), $\mu=0$, $k=\gamma=1$, $\phi=\psi=15$,
$\alpha=0.1$ and $\varphi=0.1$, with inventory bounds
$\underline{q}=-10$, $\bar{q}=10$ and lower displacement bound
$\delta_\infty=5$. We discretise the horizon on a uniform grid of
$N=1000$ time steps and average our Monte Carlo estimates over $M=3000$
independent paths. The coefficients $A(t)$, $b_0(t)$, $b_1(t)$ in
\eqref{eq:ABC-system} are pre-computed numerically on a fine auxiliary
grid.

\paragraph{Performance criterion.}
For each path we record the realised value of the objective
\eqref{eq:objective} together with the time-averaged spread
$\bar{s} := T^{-1}\!\int_0^T(\delta^{a,*}_t+\delta^{b,*}_t)\,\dd t$.
Means and standard deviations are reported in
Table~\ref{tab:perf-summary}. The ordering matches the prediction of
\cite{barucci2025market}: Full Information delivers the highest mean
objective, Partial Information comes second, and the CJP benchmark trails:
\[
  J_{\mathrm{FI}} \;\ge\; J_{\mathrm{PI}} \;\ge\; J_{\mathrm{CJP}}.
\]
Standard deviations are nearly identical across the three strategies,
which says inventory risk is comparable. The time-average spread is identical across all three
policies, consistent with the analytic result
$\delta^{a,*}+\delta^{b,*}=2/k-2A(t)$. So the performance differences
come entirely through quote skew, never through spread
adjustments.

\paragraph{Bottom line.}
Three predictions of Theorems~\ref{thm:hjb}--\ref{thm:kalman} and
Propositions~\ref{prop:optimal-controls}--\ref{prop:linear-odes} are
borne out by the simulations: (i) the time-averaged spread is identical
to four decimals across FI, PI, and CJP ($\bar{s}=2.0930$ in
Table~\ref{tab:perf-summary}), matching the closed-form invariance
$\delta^{a,*}+\delta^{b,*}=2/k-2A(t)$ from
\eqref{eq:optimal-quadratic}; (ii) the ordering
$J_{\mathrm{FI}}\ge J_{\mathrm{PI}}\ge J_{\mathrm{CJP}}$ holds in mean at
baseline ($21.33$, $21.20$, $21.18$) and the FI--CJP gap widens with $q$
in Figure~\ref{fig:perf-q}, with the cost of misspecification growing
fastest in the fad-dominated regime $q\to 1$; (iii) the Kalman--Bucy
filter from Theorem~\ref{thm:kalman} closes the FI--PI portion of that
gap to within numerical noise as $q\to 1$, consistent with the
$\hat{U}_t\to U_t$ convergence visible in Figure~\ref{fig:filter}.

\subsection*{A.1 \quad Black-Scholes Call Price (Question 1)}

\begin{lstlisting}[style=pythonstyle,
  caption={Black-Scholes European call price and convexity check.}]
import numpy as np
import matplotlib.pyplot as plt
from scipy.stats import norm

np.random.seed(42)

def CallBS(S0, K, r, sigma, T):
    """Black-Scholes European call option price."""
    d1 = (np.log(S0 / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    return S0 * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)

# --- Convexity check ---
K_grid = np.linspace(35, 55, 200)
C_grid = np.array([CallBS(50, K, 0.05, 0.30, 1) for K in K_grid])
second_diff = C_grid[:-2] - 2 * C_grid[1:-1] + C_grid[2:]
print(f"Min second finite difference: {second_diff.min():.4e}")  # > 0

# --- Plot ---
fig, ax = plt.subplots(figsize=(7, 4))
ax.plot(K_grid, C_grid, linewidth=2)
ax.set_xlabel("Strike $K$"); ax.set_ylabel("$C_{BS}(K)$")
ax.set_title("Black-Scholes call price vs $K$  "
             r"($S_0=50$, $r=0.05$, $\sigma=0.30$, $T=1$)")
ax.grid(True, linestyle="--", alpha=0.5)
plt.tight_layout()
plt.savefig("fig_q1_convexity.pdf", bbox_inches="tight")
\end{lstlisting}

\subsection*{A.2 \quad GBM Path Simulation}

\begin{lstlisting}[style=pythonstyle,
  caption={Exact log-normal GBM path simulation under the risk-neutral measure.}]
def simulate_paths(S0, r, sigma, T, n_steps, M):
    """
    Simulate M GBM paths on a uniform grid of n_steps+1 dates.
    Returns an (M, n_steps+1) array of asset prices.
    """
    dt = T / n_steps
    Z  = np.random.randn(M, n_steps)          # i.i.d. N(0,1) increments
    S  = np.zeros((M, n_steps + 1))
    S[:, 0] = S0
    for t in range(1, n_steps + 1):
        S[:, t] = S[:, t-1] * np.exp(
            (r - 0.5 * sigma**2) * dt + sigma * np.sqrt(dt) * Z[:, t-1]
        )
    return S
\end{lstlisting}

    -------
    price : float   discounted MC estimate
    std   : float   sample std of discounted payoffs
    """
    S = simulate_paths(S0, r, sigma, T, n_steps, M)

    if method == "arithmetic":
        A = np.mean(S, axis=1)
    elif method == "geometric":
        A = np.exp(np.mean(np.log(S), axis=1))
    else:
        raise ValueError("method must be 'arithmetic' or 'geometric'")

    disc_payoff = np.exp(-r * T) * np.maximum(A - K, 0)
    return disc_payoff.mean(), disc_payoff.std(ddof=1)


def confidence_interval_95(mean, std, M):
    """Return (lower, upper, width) of the 95% MC confidence interval."""
    hw = 1.96 * std / np.sqrt(M)
    return mean - hw, mean + hw, 2 * hw
\end{lstlisting}
